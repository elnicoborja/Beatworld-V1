      const interval = setInterval(() => {
        count++;
        setStarsRevealed(count);
        if (count === maxStars) {
          clearInterval(interval);
          setTimeout(() => setPhase(2), 400); // Small pause before typing
        }
      }, 150);
      return () => clearInterval(interval);

      const interval = setInterval(() => {
        setDisplayedText(fullReviewText.slice(0, idx + 1));
        idx++;
        if (idx === fullReviewText.length) {
          clearInterval(interval);
          setTimeout(() => setPhase(3), 800); // Pause before clout award

      return () => clearInterval(interval);

          0%, 100% { opacity: 0.3; }
          50% { opacity: 1; }

          display: inline-block;

          100% { transform: rotateY(360deg); }
        }

                  +500 CLOUT

              </svg>

              <CoinSprite />

              {[...Array(8)].map((_, i) => {
                const angle = (i * 45) * (Math.PI / 180);
                const distance = 40;
                const tx = Math.cos(angle) * distance;
                const ty = Math.sin(angle) * distance;
                return (
                  <div 
                    key={`p-${i}`}
                    className="absolute w-[6px] h-[6px] bg-[#ffee00] border border-[#111] anim-particle pointer-events-none"
                    style={{
                      left: '50%', top: '50%',
                      marginLeft: '-3px', marginTop: '-3px',
                      '--tx': `${tx}px`, '--ty': `${ty}px`
                    } as React.CSSProperties}
                  />
                );
              })}
            </div>
          )}
        </div>

        {/* Unlock Badge (Shows in Phase 3) */}
        {phase === 3 && (
          <div className="bg-[#44cc00] border-[2px] border-[#111] shadow-[3px_3px_0_#111] px-[20px] py-[16px] mb-[40px] anim-bounce flex items-center justify-center w-full max-w-[400px]">

              🔓 MIAMI UNLOCKED!

        @keyframes fadeIn {
          from { opacity: 0; transform: translateY(10px); }
