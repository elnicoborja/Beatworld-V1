import React, { useState, useEffect, useMemo } from 'react';

// BEATWORLD Strict Color Palette
const COLORS = {
  magenta: '#ff00ff',
  cyan: '#00ffff',
  yellow: '#ffff00',
  electricBlue: '#0050ff',
  red: '#ff0000',
  green: '#00ff00',
  hotPink: '#ff00c8',
  orange: '#ff8800',
  navy: '#0a0a1a',
  midNavy: '#111133',
  white: '#e0e0ff',
  gray: '#444466',
  black: '#000000',
};

const CYCLE_COLORS = [COLORS.magenta, COLORS.cyan, COLORS.yellow, COLORS.orange];

export default function PerformanceScreen() {
  const [isPlaying, setIsPlaying] = useState(true);
  const [vuLevels, setVuLevels] = useState<number[]>(Array(16).fill(1));

  // Simulate Performance Duration
  useEffect(() => {
    const timer = setTimeout(() => {
      setIsPlaying(false);
    }, 5000); // 5 seconds for demo

    return () => clearTimeout(timer);
  }, []);

  // VU Meter Animation
  useEffect(() => {
    let interval: NodeJS.Timeout;
    if (isPlaying) {
      interval = setInterval(() => {
        setVuLevels(Array(16).fill(0).map(() => Math.floor(Math.random() * 4) + 1));
      }, 150); // Fast pulse
    } else {
      setVuLevels(Array(16).fill(1)); // Reset to low on end
    }
    return () => clearInterval(interval);
  }, [isPlaying]);

  // Generate Crowd (Memoized so it doesn't reshuffle on every render)
  const crowd = useMemo(() => {
    const rows = [];
    const numRows = 4;
    const cols = 35;
    for (let r = 0; r < numRows; r++) {
      const rowY = 160 + r * 15;
      for (let c = 0; c < cols; c++) {
        const x = c * 14 - 20 + (r % 2) * 6; // Staggered rows
        const delay = Math.random() * 0.5;
        const color = CYCLE_COLORS[Math.floor(Math.random() * CYCLE_COLORS.length)];
        rows.push({ id: `head-${r}-${c}`, x, y: rowY, delay, color });
      }
    }
    return rows;
  }, []);

  // Generate Confetti
  const confetti = useMemo(() => {
    return Array(40).fill(0).map((_, i) => ({
      id: i,
      color: CYCLE_COLORS[Math.floor(Math.random() * CYCLE_COLORS.length)],
      tx: (Math.random() - 0.5) * 400, // -200px to 200px
      ty: (Math.random() - 1) * 300 - 50, // Fly up between -50px and -350px
      delay: Math.random() * 0.2,
    }));
  }, []);

  return (
    <div className="relative w-screen h-screen bg-[#0a0a1a] text-[#e0e0ff] uppercase overflow-hidden font-press-start select-none flex flex-col items-center justify-center">
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Press+Start+2P&display=swap');
        
        .font-press-start {
          font-family: 'Press Start 2P', monospace;
        }

        /* Overlays */
        .scanlines {
          background: repeating-linear-gradient(
            to bottom,
            rgba(0, 0, 0, 0) 0px,
            rgba(0, 0, 0, 0) 2px,
            rgba(0, 0, 0, 0.1) 2px,
            rgba(0, 0, 0, 0.1) 4px
          );
          pointer-events: none;
        }

        .grid-overlay {
          background-image: 
            linear-gradient(rgba(255, 0, 255, 0.03) 1px, transparent 1px),
            linear-gradient(90deg, rgba(255, 0, 255, 0.03) 1px, transparent 1px);
          background-size: 32px 32px;
          pointer-events: none;
        }

        .pixel-shadow { box-shadow: 3px 3px 0px #000; }
        .text-glow-magenta { text-shadow: 0 0 8px #ff00ff; }
        .text-glow-cyan { text-shadow: 0 0 8px #00ffff; }
        
        /* Animations */
        @keyframes crowdBob {
          0%, 100% { transform: translateY(0px); }
          50% { transform: translateY(-2px); }
        }
        .anim-crowd {
          animation: crowdBob 0.4s infinite steps(2);
        }

        @keyframes lightBeamFlash {
          0%, 100% { opacity: 0.6; }
          50% { opacity: 0.1; }
        }
        .anim-beam-1 { animation: lightBeamFlash 0.3s infinite steps(2); }
        .anim-beam-2 { animation: lightBeamFlash 0.3s infinite steps(2) 0.1s; }
        .anim-beam-3 { animation: lightBeamFlash 0.3s infinite steps(2) 0.2s; }
        .anim-beam-4 { animation: lightBeamFlash 0.3s infinite steps(2) 0.15s; }

        @keyframes stageGlowPulse {
          0% { box-shadow: 0 -20px 40px -10px #ff00ff; border-top-color: #ff00ff; }
          33% { box-shadow: 0 -20px 40px -10px #00ffff; border-top-color: #00ffff; }
          66% { box-shadow: 0 -20px 40px -10px #ffff00; border-top-color: #ffff00; }
          100% { box-shadow: 0 -20px 40px -10px #ff8800; border-top-color: #ff8800; }
        }
        .anim-stage-glow {
          animation: stageGlowPulse 1.2s infinite steps(4);
        }

        @keyframes textBlink {
          0%, 100% { opacity: 1; }
          50% { opacity: 0; }
        }
        .anim-blink {
          animation: textBlink 1s infinite steps(2);
        }

        @keyframes confettiFly {
          0% { transform: translate(0, 0) scale(1); opacity: 1; }
          100% { transform: translate(var(--tx), var(--ty)) scale(0.5); opacity: 0; }
        }
        .anim-confetti {
          animation: confettiFly 1.5s ease-out forwards;
        }

        /* Light Beam Polygon */
        .light-beam {
          clip-path: polygon(calc(50% - 30px) 0, calc(50% + 30px) 0, calc(50% + 100px) 100%, calc(50% - 100px) 100%);
        }
      `}</style>

      {/* BACKGROUND SVG VENUE */}
      <div className="absolute inset-0 z-0 bg-[#0d0d22] flex items-end justify-center">
        <svg width="100%" height="100%" viewBox="0 0 400 300" preserveAspectRatio="xMidYMax slice" style={{ imageRendering: 'pixelated' }}>
          {/* Back Wall Signage Silhouette */}
          <rect x="150" y="40" width="100" height="40" fill="#111133" />
          <rect x="160" y="50" width="80" height="20" fill="#0a0a1a" />
          
          {/* Lighting Rig Bar */}
          <rect x="0" y="20" width="400" height="10" fill="#0a0a1a" stroke="#444466" strokeWidth="2" />
          {/* Downward Cones (Lights) */}
          <polygon points="40,30 60,30 50,50" fill="#ff00ff" />
          <polygon points="140,30 160,30 150,50" fill="#00ffff" />
          <polygon points="240,30 260,30 250,50" fill="#ffff00" />
          <polygon points="340,30 360,30 350,50" fill="#ff8800" />

          {/* Crowd */}
          <g>
            {crowd.map((person) => (
              <g 
                key={person.id} 
                className={isPlaying ? 'anim-crowd' : ''} 
                style={{ animationDelay: `${person.delay}s` }}
              >
                {/* Head Box */}
                <rect x={person.x} y={person.y} width="12" height="8" fill={person.color} opacity="0.8" />
                {/* Hair/Detail */}
                <rect x={person.x + 2} y={person.y - 2} width="8" height="4" fill="#0a0a1a" />
              </g>
            ))}
          </g>

          {/* Speaker Stacks */}
          <g transform="translate(10, 140)">
            <rect x="0" y="0" width="40" height="80" fill="#0a0a1a" stroke="#444466" strokeWidth="2" />
            <circle cx="20" cy="20" r="12" fill="#111133" />
            <circle cx="20" cy="60" r="12" fill="#111133" />
          </g>
          <g transform="translate(350, 140)">
            <rect x="0" y="0" width="40" height="80" fill="#0a0a1a" stroke="#444466" strokeWidth="2" />
            <circle cx="20" cy="20" r="12" fill="#111133" />
            <circle cx="20" cy="60" r="12" fill="#111133" />
          </g>
        </svg>
      </div>

      {/* STAGE FLOOR (Bottom 30%) */}
      <div 
        className={`absolute bottom-0 w-full h-[30%] bg-[#0a0a1a] border-t-2 z-10 flex flex-col ${isPlaying ? 'anim-stage-glow' : 'border-[#444466] shadow-[0_-20px_40px_-10px_#444466]'}`}
      >
        {/* Floorboards */}
        <div className="w-full h-[4px] bg-[#111133] mt-2 opacity-50" />
        <div className="w-full h-[4px] bg-[#111133] mt-4 opacity-40" />
        <div className="w-full h-[4px] bg-[#111133] mt-6 opacity-30" />
        <div className="w-full h-[4px] bg-[#111133] mt-8 opacity-20" />
      </div>

      {/* LIGHT BEAMS (CSS Keyframes) */}
      {isPlaying && (
        <div className="absolute top-[30px] left-0 w-full h-[40vh] z-10 flex justify-between px-[5%] pointer-events-none mix-blend-screen">
          <div className="w-[100px] h-full bg-[#ff00ff] light-beam anim-beam-1 transform origin-top rotate-[15deg]" />
          <div className="w-[100px] h-full bg-[#00ffff] light-beam anim-beam-2 transform origin-top rotate-[5deg]" />
          <div className="w-[100px] h-full bg-[#ffff00] light-beam anim-beam-3 transform origin-top rotate-[-5deg]" />
          <div className="w-[100px] h-full bg-[#ff8800] light-beam anim-beam-4 transform origin-top rotate-[-15deg]" />
        </div>
      )}

      {/* OVERLAYS */}
      <div className="absolute inset-0 z-20 grid-overlay" />
      <div className="absolute inset-0 z-30 scanlines" />

      {/* CENTER HUD */}
      <div className="relative z-40 flex flex-col items-center gap-[24px]">
        {/* Venue Panel */}
        <div className="bg-[#111133]/aa border-[2px] border-[#ff00ff] p-[16px] flex flex-col items-center gap-[8px] pixel-shadow backdrop-blur-sm">
          <h1 className="text-[#ff00ff] text-[11px] text-glow-magenta m-0 text-center">
            🎪 MADISON SQUARE GARDEN
          </h1>
          <p className="text-[#00ffff] text-[7px] m-0 text-center tracking-widest text-glow-cyan">
            NEW YORK CITY · BOOM BAP
          </p>
        </div>

        {/* VU Meter Row */}
        <div className="bg-[#111133] border border-[#444466] p-[8px] flex gap-[4px] items-end h-[60px] pixel-shadow">
          {vuLevels.map((level, i) => {
            const heightMap = { 1: '25%', 2: '50%', 3: '75%', 4: '100%' };
            const h = heightMap[level as keyof typeof heightMap];
            
            // Color based on height: 1=magenta, 2=yellow, 3/4=red
            let color = COLORS.magenta;
            if (level === 2) color = COLORS.yellow;
            if (level >= 3) color = COLORS.red;
            
            return (
              <div 
                key={`vu-${i}`} 
                className="w-[8px] transition-all duration-75 ease-linear"
                style={{ 
                  height: h, 
                  backgroundColor: color,
                  boxShadow: `0 0 8px ${color}`
                }}
              />
            );
          })}
        </div>
      </div>

      {/* PERFORMANCE END STATE */}
      {!isPlaying && (
        <div className="absolute inset-0 z-50 flex flex-col items-center justify-center bg-[#0a0a1a]/80 backdrop-blur-sm">
          
          <h2 className="text-[#ffff00] text-[24px] md:text-[36px] text-center mb-[32px] anim-blink drop-shadow-[0_0_12px_#ffff00] leading-tight">
            PERFORMANCE<br/>COMPLETE!
          </h2>
          
          <button className="bg-[#ff00ff] text-[#0a0a1a] border-[2px] border-[#ff00ff] px-[24px] py-[16px] text-[10px] uppercase pixel-shadow hover:brightness-110 active:translate-y-[2px] active:translate-x-[2px] active:shadow-[1px_1px_0px_#000] transition-none outline-none focus:ring-2 focus:ring-[#ffff00]">
            SEE REVIEWS
          </button>

          {/* Confetti Burst */}
          <div className="absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 pointer-events-none">
            {confetti.map((c) => (
              <div
                key={`confetti-${c.id}`}
                className="absolute w-[4px] h-[4px] anim-confetti"
                style={{
                  backgroundColor: c.color,
                  boxShadow: `0 0 4px ${c.color}`,
                  // @ts-ignore - Custom CSS variables for animation
                  '--tx': `${c.tx}px`,
                  '--ty': `${c.ty}px`,
                  animationDelay: `${c.delay}s`
                }}
              />
            ))}
          </div>
        </div>
      )}

    </div>
  );
}
